JavaScript 基础语法

每段代码下面那个黄底框是页面运行时真实打印出来的结果，改上面的代码就能看到变化。

1. 变量：let / const / var

let score = 60;        // 可重新赋值
const MAX = 100;       // 常量，不能再赋值
score = 90;
console.log(score, MAX);

2. 数据类型与 typeof

const vals = ['字符串', 42, 3.14, true, null, undefined, {k: 1}, [1, 2]];
vals.forEach(v => console.log(typeof v, JSON.stringify(v)));

3. 类型转换与「相等」的坑

console.log('1' + 1);          // 字符串拼接 -> '11'
console.log('1' - 1);          // 减法会转数字 -> 0
console.log(Number('abc'));    // NaN
console.log('1' == 1, '1' === 1);   // true / false
console.log(NaN === NaN);      // false

4. 字符串模板与常用方法

const name = 'zhoumo';
const greet = `你好，${name.length > 3 ? '长名字' : '短名字'}！`;
console.log(greet);
console.log('  a,b,c  '.trim().split(','));
console.log('Vue3 组件'.includes('组件'));

5. 条件与三元

function level(n) {
  if (n >= 90) return 'A';
  else if (n >= 60) return 'B';
  else return 'C';
}
console.log(level(95), level(72), level(30));
console.log(true ? '真' : '假');

6. 函数：声明式 / 表达式 / 箭头 / 默认参数

function add(a, b) { return a + b; }
const sub = function (a, b) { return a - b; };
const mul = (a, b) => a * b;
const hello = (who = '同学') => `Hi, ${who}`;
console.log(add(2, 3), sub(9, 4), mul(3, 4), hello());

7. 对象与解构

const user = { name: '小丽', age: 20, vip: true };
const { name, age, vip } = user;
user.city = '长沙';                       // 动态加属性
console.log(name, age, vip, user.city);
console.log(Object.keys(user), Object.values(user));

8. 练习

练习 1：写一个函数 bmi(体重kg, 身高m)，<18.5 偏瘦、<24 正常、否则偏胖 
const bmi = (kg, m) => {
  const v = kg / (m * m);
  return v < 18.5 ? '偏瘦' : v < 24 ? '正常' : '偏胖';
};
console.log(bmi(70, 1.75));
练习 2：把 "2026-10-12" 拆成年月日三个变量 
const [y, m, d] = '2026-10-12'.split('-');
console.log(y, m, d);
