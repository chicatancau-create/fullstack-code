/* 极坐标圆盘（朝 +Y，中心退化为首环扇形） */
function discGeometry(radius, rings, sectors, yFn, nFn, cFn) {
  const S = sectors, pos = [], nor = [], col = [], uv = [], idx = [];
  const tmpC = new THREE.Color(), tmpN = new THREE.Vector3();
  for (let i = 0; i <= rings; i++) {
    const r = radius * i / rings;
    for (let j = 0; j < S; j++) {
      const a = j / S * Math.PI * 2, cx = Math.cos(a) * r, cz = Math.sin(a) * r;
      const y = yFn ? yFn(cx, cz, r) : 0;
      pos.push(cx, y, cz);
      uv.push(cx * 1.0, cz * 1.0);
      if (nFn) { nFn(cx, cz, tmpN); nor.push(tmpN.x, tmpN.y, tmpN.z); }
      if (cFn) { cFn(cx, y, cz, tmpC); col.push(tmpC.r, tmpC.g, tmpC.b); }
    }
  }
  for (let i = 0; i < rings; i++) for (let j = 0; j < S; j++) {
    const j2 = (j + 1) % S;
    const a = i * S + j, b = i * S + j2, c = (i + 1) * S + j, d = (i + 1) * S + j2;
    if (i > 0) idx.push(a, b, d);
    idx.push(a, d, c);
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
  if (nFn) g.setAttribute('normal', new THREE.Float32BufferAttribute(nor, 3));
  if (cFn) g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
  g.setIndex(idx);
  if (!nFn) g.computeVertexNormals();
  g.computeBoundingSphere();
  return g;
}

/* ---------------------------------------------------------------- 沙地色板 */
const SAND_DRY = new THREE.Color(0xdcae6a), SAND_WET = new THREE.Color(0xa87b41),
      SAND_SUB = new THREE.Color(0xbda06a), MUD = new THREE.Color(0x6d6350),
      ROCK = new THREE.Color(0x4a4640), MOSS = new THREE.Color(0x7d8a56);
function terrainColor(x, h, z, out) {
  const v = fbm(x * 5.2 + 11, z * 5.2 - 4, 3, 61) * 0.5 + 0.5;
  if (h > WATER_Y + 0.115) {
    out.copy(SAND_DRY).lerp(MOSS, smoothstep(0.42, 0.60, h) * 0.5 * v);
    out.lerp(SAND_WET, 0.10 + v * 0.12);
  } else if (h > WATER_Y - 0.03) {
    out.copy(SAND_WET).lerp(SAND_DRY, smoothstep(WATER_Y - 0.03, WATER_Y + 0.10, h));
  } else {
    out.copy(SAND_SUB).lerp(MUD, smoothstep(WATER_Y - 0.05, -0.22, h));
    out.lerp(ROCK, smoothstep(-0.24, -0.40, h) * (0.55 + 0.45 * v));
  }
  out.multiplyScalar(0.88 + v * 0.24);
  return out;
}

const terrainGeo = discGeometry(
  CORE, 96, 168,
  (x, z) => terrainH(x, z),
  (x, z, n) => terrainN(x, z, n),
  (x, y, z, c) => terrainColor(x, y, z, c)
);
const sandMat = new THREE.MeshStandardMaterial({
  vertexColors: true, roughness: 0.92, metalness: 0.0,
  normalMap: TEX.sandNormal, normalScale: new THREE.Vector2(0.85, 0.85)
});
sandMat.normalMap.repeat.set(7, 7);
applyUnderwater(sandMat, { caustic: 1.0, wetRough: 0.22, tag: 'sand' });
const terrain = new THREE.Mesh(terrainGeo, sandMat);
terrain.castShadow = true; terrain.receiveShadow = true;
island.add(terrain);

/* 截面侧壁：世界高度对齐的沉积层 */
function buildCoreWall(radius, sectors) {
  const pos = [], nor = [], uv = [], idx = [];
  const STEPS = 26;
  for (let j = 0; j <= sectors; j++) {
    const a = j / sectors * Math.PI * 2;
    const cx = Math.cos(a), cz = Math.sin(a);
    const top = Math.max(terrainH(cx * radius, cz * radius), Y_BOT + 0.02);
    for (let k = 0; k <= STEPS; k++) {
      const y = mix(top, Y_BOT, k / STEPS);
      pos.push(cx * radius, y, cz * radius);
      nor.push(cx, 0, cz);
      uv.push(j / sectors, (y - Y_BOT) / (Y_TOP - Y_BOT));
    }
  }
  const COL = STEPS + 1;
  for (let j = 0; j < sectors; j++) for (let k = 0; k < STEPS; k++) {
    const a = j * COL + k, b = a + COL, c = a + 1, d = b + 1;
    idx.push(a, b, d, a, d, c);
  }

  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
