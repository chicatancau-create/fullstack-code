      float h = 0.0;
      h += 0.0046*sin(dot(p,W1)+t*1.15);
      h += 0.0034*sin(dot(p,W2)+t*1.52);
      h += 0.0024*sin(dot(p,W3)+t*2.17);
      h += 0.0015*sin(dot(p,W4)+t*2.93);

      d += W3*0.0024*cos(dot(p,W3)+t*2.17);
      d += W4*0.0015*cos(dot(p,W4)+t*2.93);
      return d;
    }
    void main(){
      vec3 pos = position;
      float rho = length(pos.xz)/uR;
      float fade = smoothstep(1.0, 0.86, rho);

      float detail = 0.0016*fbm2(pos.xz*3.4 + vec2(t*0.16, t*0.09));
      pos.y = uWaterY + (wh(pos.xz, t) + detail) * fade * uWindI;
      vW = pos.y; vRho = rho;

      gl_Position = projectionMatrix*viewMatrix*wp;
