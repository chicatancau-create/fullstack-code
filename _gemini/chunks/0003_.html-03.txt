/* ---------------------------------------------------------------- 噪声 */
function imul(a, b) { return Math.imul(a, b) | 0; }
function hash2i(ix, iy, seed) {
  let n = imul(ix, 374761393) + imul(iy, 668265263) + imul(seed + 1, 1442695041);
  n = imul(n ^ (n >>> 13), 1274126177);
  n ^= n >>> 16;
  return (n >>> 0) / 4294967295;
}
function smoothstep(a, b, x) {
  const t = Math.min(1, Math.max(0, (x - a) / (b - a)));
  return t * t * (3 - 2 * t);
}
const mix = (a, b, t) => a + (b - a) * t;

function vnoise(x, y, seed = 0) {
  const xi = Math.floor(x), yi = Math.floor(y);
  const xf = x - xi, yf = y - yi;
  const u = xf * xf * (3 - 2 * xf), v = yf * yf * (3 - 2 * yf);
  const a = hash2i(xi, yi, seed), b = hash2i(xi + 1, yi, seed);
  const c = hash2i(xi, yi + 1, seed), d = hash2i(xi + 1, yi + 1, seed);
  return mix(mix(a, b, u), mix(c, d, u), v);
}
function fbm(x, y, oct = 4, seed = 0, gain = 0.5, lac = 2.03) {
  let s = 0, amp = 0.5, f = 1, norm = 0;
  for (let i = 0; i < oct; i++) { s += amp * (vnoise(x * f, y * f, seed + i * 17) * 2 - 1); norm += amp; amp *= gain; f *= lac; }
  return s / norm;           // -1..1
}
