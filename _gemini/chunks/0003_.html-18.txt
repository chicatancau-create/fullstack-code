function fishGeometry(len = 0.052) {
  const body = new THREE.SphereGeometry(len, 11, 8);
  body.scale(1.0, 0.46, 0.36);
  const tail = new THREE.ConeGeometry(len * 0.62, len * 0.8, 4, 1, true);
  tail.rotateZ(Math.PI / 2); tail.scale(0.35, 1.0, 1.5);
  tail.translate(-len * 1.22, 0, 0);
  const dorsal = new THREE.ConeGeometry(len * 0.30, len * 0.42, 3, 1);
  dorsal.rotateZ(-Math.PI / 2); dorsal.scale(1, 0.16, 0.5);
  dorsal.translate(-len * 0.1, len * 0.26, 0);
  const g = merge([np(body), np(tail), np(dorsal)]);
  const p = g.attributes.position;
  const f = new Float32Array(p.count), c = new Float32Array(p.count * 3).fill(1);
  for (let i = 0; i < p.count; i++) f[i] = smoothstep(0.1, -0.9, p.getX(i) / len);
  g.setAttribute('aFlex', new THREE.Float32BufferAttribute(f, 1));
  g.setAttribute('color', new THREE.Float32BufferAttribute(c, 3));
  if (!g.attributes.uv) g.setAttribute('uv', new THREE.Float32BufferAttribute(p.count * 2, 2));
  return g;
}
