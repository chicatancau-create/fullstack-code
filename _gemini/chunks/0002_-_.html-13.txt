const koiColors = [[0xff6b35, 0xfff4e0], [0xfff4e0, 0xff6b35], [0xffb020, 0xfff4e0], [0xff6b35, 0xff6b35], [0xfff4e0, 0xffb020], [0xe8590c, 0xfff4e0]];
for (let i = 0; i < 6; i++) {
  const g = new THREE.Group();
  const [c1, c2] = koiColors[i];
  const m1 = new THREE.MeshStandardMaterial({ color: c1, roughness: 0.5 });
  const m2 = new THREE.MeshStandardMaterial({ color: c2, roughness: 0.5 });
  const head = new THREE.Mesh(boxGeo, m1); head.scale.set(0.42, 0.28, 0.5); head.position.z = 0.42;
  const body = new THREE.Mesh(boxGeo, i % 2 ? m2 : m1); body.scale.set(0.48, 0.32, 0.62);
  const tail = new THREE.Mesh(boxGeo, m2); tail.scale.set(0.3, 0.22, 0.42); tail.position.z = -0.52;
  const fin = new THREE.Mesh(boxGeo, m2); fin.scale.set(0.56, 0.06, 0.24); fin.position.set(0, -0.1, 0.1);
  g.add(head, body, tail, fin);
  g.traverse(o => { if (o.isMesh) { o.castShadow = false; } });
  scene.add(g);
  kois.push({ g, tail, r: 2 + (i % 3) * 1.35, sp: rr(0.25, 0.5) * (i % 2 ? 1 : -1), ph: rr(0, 9), y: rr(-1.6, -1.0) });
}
function updateKoi(t) {
  for (const k of kois) {
    const a = t * k.sp + k.ph;
    const x = 9 + Math.cos(a) * k.r, z = 9 + Math.sin(a) * k.r;
    k.g.position.set(x, k.y + Math.sin(t * 0.8 + k.ph) * 0.18, z);
    const vx = -Math.sin(a) * k.sp, vz = Math.cos(a) * k.sp;
    k.g.rotation.y = Math.atan2(vx, vz);
    k.tail.rotation.y = Math.sin(t * 6 + k.ph) * 0.45;
    k.g.rotation.z = Math.sin(t * 2 + k.ph) * 0.06;
  }
}

/* ============ 蝴蝶（细单元） ============ */
const butterflies = [];
const wingGeo = new THREE.BoxGeometry(0.3, 0.02, 0.22); wingGeo.translate(0.16, 0, 0);
const bflyCols = [0x22d3ee, 0xf06595, 0xffc078, 0xb197fc, 0x69db7c, 0xff8787, 0x74c0fc, 0xf783ac];
const bflyCenters = beds.map(b => [b[0], b[1]]).concat([[-15, 5], [15, -6]]);
for (let i = 0; i < 8; i++) {
  const g = new THREE.Group();
  const mat = new THREE.MeshStandardMaterial({ color: bflyCols[i], roughness: 0.5, side: THREE.DoubleSide });
  const bodyM = new THREE.Mesh(boxGeo, new THREE.MeshStandardMaterial({ color: 0x2b2b2b }));
  bodyM.scale.set(0.07, 0.07, 0.3);
  const wl = new THREE.Mesh(wingGeo, mat), wr = new THREE.Mesh(wingGeo, mat);
  wr.scale.x = -1;
  g.add(bodyM, wl, wr); scene.add(g);
  const c = bflyCenters[i % bflyCenters.length];
  butterflies.push({ g, wl, wr, cx: c[0], cz: c[1], ph: rr(0, 9), sp: rr(0.5, 1), r: rr(1.4, 2.6) });
}
function updateButterflies(t) {
  for (const b of butterflies) {
    const a = t * b.sp + b.ph;
    const x = b.cx + Math.cos(a) * b.r + Math.sin(a * 2.3) * 0.5;
    const z = b.cz + Math.sin(a * 0.9) * b.r;
    const y = 1.3 + Math.sin(a * 1.7) * 0.55 + Math.sin(t * 3 + b.ph) * 0.12;
    b.g.position.set(x, y, z);
    b.g.rotation.y = Math.atan2(-Math.sin(a) * b.sp, Math.cos(a * 0.9) * b.sp * 0.9);
    const flap = Math.sin(t * 18 + b.ph) * 0.75 + 0.25;
    b.wl.rotation.z = flap; b.wr.rotation.z = -flap;
  }
}

/* ============ 仙鹤：展翅 ============ */
function craneModel() {
  const g = new THREE.Group();
  const CV = [];
  const cv = (x, y, z, sx, sy, sz, c) => CV.push({ x, y, z, sx, sy, sz, c });
  const WH = 0xf8f6f0, BK = 0x22262b;
  cv(0, 0, 0, 0.9, 0.7, 1.5, WH);
  cv(0, 0.15, -0.8, 0.7, 0.5, 0.6, BK);
  cv(0, 0.35, 0.7, 0.3, 0.5, 0.3, WH);
  cv(0, 0.75, 0.95, 0.24, 0.5, 0.24, WH);
  cv(0, 1.15, 1.1, 0.22, 0.45, 0.22, WH);
  cv(0, 1.5, 1.2, 0.3, 0.3, 0.4, WH);
  cv(0, 1.68, 1.22, 0.16, 0.12, 0.2, 0xe03131);
  cv(0, 1.47, 1.55, 0.12, 0.1, 0.55, 0x4a4f55);
  cv(-0.28, -0.75, 0.1, 0.09, 1.0, 0.09, BK);
  cv(0.28, -0.75, -0.1, 0.09, 1.0, 0.09, BK);
  g.add(bakeVoxels(CV, stdMat));
  const wings = [];
  for (const s of [-1, 1]) {
    const wg = new THREE.Group(); wg.position.set(s * 0.42, 0.28, 0);
    const WV = [];
    for (let k = 0; k < 5; k++) {
      WV.push({ x: s * (0.35 + k * 0.42), y: k * 0.06, z: -k * 0.06, sx: 0.46, sy: 0.12, sz: 0.9 - k * 0.1, c: k >= 3 ? BK : WH });
    }
    wg.add(bakeVoxels(WV, stdMat));
    g.add(wg); wings.push(wg);
  }
  return { g, wings };
}
