const pondC = { x: 9, z: 9 };
function pondSigned(x, z) {
  const dx = x - pondC.x, dz = z - pondC.z;
  const ang = Math.atan2(dz, dx);
  const rp = 6.2 + 1.1 * Math.sin(ang * 3 + 0.8) + 0.5 * Math.sin(ang * 5 + 2);
  return Math.hypot(dx, dz) - rp;
}
