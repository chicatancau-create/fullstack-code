/* ============ 相机控制：阻尼环绕 / 缩放 / 平移 ============ */
const INIT = { theta: 0.65, phi: 1.12, dist: 62, tx: 0, ty: 5, tz: 0 };
const goal = { ...INIT }, cur = { ...INIT };
let autoRotate = true, dragging = false;
const pointers = new Map();
let pinchD = 0;
canvas.addEventListener('contextmenu', e => e.preventDefault());
canvas.addEventListener('pointerdown', e => {
  canvas.setPointerCapture(e.pointerId);
  pointers.set(e.pointerId, { x: e.clientX, y: e.clientY, b: e.button, shift: e.shiftKey });
  if (pointers.size === 2) {
    const [a, b] = [...pointers.values()];
    pinchD = Math.hypot(a.x - b.x, a.y - b.y);
  }
  dragging = true; canvas.classList.add('dragging');
});
canvas.addEventListener('pointermove', e => {
  const p = pointers.get(e.pointerId);
  if (!p) return;
  const dx = e.clientX - p.x, dy = e.clientY - p.y;
  p.x = e.clientX; p.y = e.clientY;
  if (pointers.size === 1) {
    if (p.b === 2 || p.shift) pan(dx, dy);
    else { goal.theta -= dx * 0.005; goal.phi = clamp(goal.phi - dy * 0.005, 0.12, 2.15); }
  } else if (pointers.size === 2) {
    const [a, b] = [...pointers.values()];
    const d = Math.hypot(a.x - b.x, a.y - b.y);
    if (pinchD > 0) goal.dist = clamp(goal.dist * pinchD / d, 14, 160);
    pinchD = d;
    pan(dx * 0.5, dy * 0.5);
  }
});
function endPointer(e) {
  pointers.delete(e.pointerId);
  if (pointers.size < 2) pinchD = 0;
  if (pointers.size === 0) { dragging = false; canvas.classList.remove('dragging'); }
}
canvas.addEventListener('pointerup', endPointer);
canvas.addEventListener('pointercancel', endPointer);
canvas.addEventListener('wheel', e => {
  e.preventDefault();
  goal.dist = clamp(goal.dist * Math.exp(e.deltaY * 0.0012), 14, 160);
}, { passive: false });
function pan(dx, dy) {
  const scale = cur.dist * 0.0011;
  const sin = Math.sin(cur.theta), cos = Math.cos(cur.theta);
  goal.tx = clamp(goal.tx - (dx * cos) * scale + (dy * sin) * scale * 0.6, -35, 35);
  goal.tz = clamp(goal.tz + (dx * sin) * scale + (dy * cos) * scale * 0.6, -35, 35);
  goal.ty = clamp(goal.ty + dy * scale * 0.8, -14, 30);
}
function updateCamera(dt) {
  if (autoRotate && !dragging) goal.theta += dt * 0.05;
  const k = 1 - Math.exp(-8 * dt);
  for (const key of ['theta', 'phi', 'dist', 'tx', 'ty', 'tz']) cur[key] += (goal[key] - cur[key]) * k;
  const comp = innerWidth < 760 ? clamp(0.95 / (innerWidth / innerHeight), 1, 2.2) : 1;
  const dist = cur.dist * comp;
  const sp = Math.sin(cur.phi), cp = Math.cos(cur.phi);
  camera.position.set(
    cur.tx + dist * sp * Math.sin(cur.theta),
    cur.ty + dist * cp,
    cur.tz + dist * sp * Math.cos(cur.theta)
  );
  camera.lookAt(cur.tx, cur.ty, cur.tz);
}
