              )}
              {jobs.length === 0 ? <p className="muted">暂无任务。</p> : (

                </table>
              )}
              <h3>通知</h3>
              {notifications.length === 0 ? <p className="muted">暂无通知。</p> : (
                <ul className="list">
                  {notifications.map((item) => <li key={item.id} className="item"><span>{item.message}</span></li>)}
                </ul>
              )}
            </div>
          )}
