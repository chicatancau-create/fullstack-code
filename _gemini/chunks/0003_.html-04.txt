  for (let i = 0; i < oct; i++) { const n = 1 - Math.abs(vnoise(x * f, y * f, seed + i * 31) * 2 - 1); s += amp * n; norm += amp; amp *= 0.5; f *= 2.07; }
  return s / norm;           // 0..1
}

/* ---------------------------------------------------------------- 程序化贴图 */
function canvasTex(size, height = size, draw, { srgb = true, repeat = [1, 1], aniso = 8 } = {}) {
  const cv = document.createElement('canvas');
  cv.width = size; cv.height = height;
  draw(cv.getContext('2d'), size, height);
  const t = new THREE.CanvasTexture(cv);
  t.wrapS = t.wrapT = THREE.RepeatWrapping;
  t.repeat.set(repeat[0], repeat[1]);
  t.colorSpace = srgb ? THREE.SRGBColorSpace : THREE.NoColorSpace;
  t.anisotropy = aniso;
  return t;
}
function normalFromHeight(w, h, hf, strength = 2.0, repeat = 1) {
  return canvasTex(w, h, (ctx) => {
    const img = ctx.createImageData(w, h), d = img.data;
    const at = (x, y) => hf((x + w) % w, (y + h) % h);
    for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
      const dx = (at(x + 1, y) - at(x - 1, y)) * strength * repeat;
      const dy = (at(x, y + 1) - at(x, y - 1)) * strength * repeat;
      let nx = -dx, ny = -dy, nz = 1;
      const l = Math.hypot(nx, ny, nz); nx /= l; ny /= l; nz /= l;
      const i = (y * w + x) * 4;
      d[i] = (nx * 0.5 + 0.5) * 255; d[i + 1] = (ny * 0.5 + 0.5) * 255;
      d[i + 2] = (nz * 0.5 + 0.5) * 255; d[i + 3] = 255;
    }
    ctx.putImageData(img, 0, 0);
  }, { srgb: false });
}
