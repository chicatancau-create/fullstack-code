			clickRight() {
				if (Number(this.currentIndex) >= this.maxPage) {
					return
				}
				this.currentIndex += 1
				this.change('next')
			},
			change(e) {
				this.$emit('input', this.currentIndex)
				this.$emit('update:modelValue', this.currentIndex)
				this.$emit('change', {
					type: e,
					current: this.currentIndex
				})
			}
		}
	}
</script>
