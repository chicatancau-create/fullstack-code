const _v1 = new THREE.Vector3(), _v2 = new THREE.Vector3(), _v3 = new THREE.Vector3();

function updateFish(dt, time) {
  const N = fish.length;
  for (let i = 0; i < N; i++) {
    const A = fish[i];
    const coh = _v1.set(0, 0, 0), ali = _v2.set(0, 0, 0), sep = _v3.set(0, 0, 0);
    let cnt = 0;
    for (let j = 0; j < N; j++) {
      if (i === j) continue;
      const B = fish[j], d = A.p.distanceTo(B.p);
      if (d < 0.24) {
        coh.add(B.p); ali.add(B.v); cnt++;
        if (d < 0.08) sep.addScaledVector(tmpV.subVectors(A.p, B.p), 0.02 / (d + 3e-3));
      }
    }
    if (cnt) { coh.multiplyScalar(1 / cnt).sub(A.p); ali.multiplyScalar(1 / cnt); }
    A.v.addScaledVector(coh, dt * 0.42).addScaledVector(ali, dt * 0.55).addScaledVector(sep, dt * 1.6);

    const wx = fbm(A.p.x * 2.6 + time * 0.22, A.p.z * 2.6, 2, 5);
    const wz = fbm(A.p.z * 2.6 - time * 0.19, A.p.x * 2.6 + 9, 2, 9);
    A.v.x += wx * dt * 0.5; A.v.z += wz * dt * 0.5; A.v.y += fbm(time * 0.3, A.ph, 2, 3) * dt * 0.3;

    const rho = Math.hypot(A.p.x, A.p.z);
    if (rho > 0.78) { const k = (rho - 0.78) * dt * 9; A.v.x -= A.p.x / rho * k; A.v.z -= A.p.z / rho * k; }
    const ceil = WATER_Y - 0.055;
    if (A.p.y > ceil) A.v.y -= (A.p.y - ceil) * dt * 8;
    const bed = terrainH(A.p.x, A.p.z) + 0.05;
    if (A.p.y < bed) A.v.y += (bed - A.p.y) * dt * 9;

    const sp = A.v.length() || 1e-5;
    const want = 0.15 * A.s;
    A.v.multiplyScalar((sp + (want - sp) * Math.min(1, dt * 2.5)) / sp);
