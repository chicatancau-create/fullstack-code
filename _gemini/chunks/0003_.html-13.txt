function stamp(g, color, flexFn) {
  const n = g.attributes.position.count, p = g.attributes.position;
  const c = new Float32Array(n * 3), f = new Float32Array(n);
  for (let i = 0; i < n; i++) {
    c[i * 3] = color.r; c[i * 3 + 1] = color.g; c[i * 3 + 2] = color.b;
    f[i] = flexFn ? flexFn(p.getX(i), p.getY(i), p.getZ(i)) : 0;
  }
  if (!g.attributes.uv) g.setAttribute('uv', new THREE.Float32BufferAttribute(n * 2, 2));
  g.setAttribute('color', new THREE.Float32BufferAttribute(c, 3));
  if (!g.attributes.aFlex) g.setAttribute('aFlex', new THREE.Float32BufferAttribute(f, 1));
  return g;
}
