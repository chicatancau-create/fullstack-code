const streamPts = [[12.5, 10.2], [16.5, 11.2], [20.5, 12.2], [25.5, 13.0]];
function segDist(px, pz, a, b) {
  const abx = b[0] - a[0], abz = b[1] - a[1];
  const t = clamp(((px - a[0]) * abx + (pz - a[1]) * abz) / (abx * abx + abz * abz), 0, 1);
  return Math.hypot(px - (a[0] + abx * t), pz - (a[1] + abz * t));
}
function streamDist(x, z) {
  let d = 1e9;
  for (let i = 0; i < streamPts.length - 1; i++) d = Math.min(d, segDist(x, z, streamPts[i], streamPts[i + 1]));
  return d;
}
const pathPts = [[0.5, 25.5], [0.5, 17], [-1.5, 11], [-4, 5], [-6.5, 0], [-8, -3.5]];
function pathDist(x, z) {
  let d = 1e9;
  for (let i = 0; i < pathPts.length - 1; i++) d = Math.min(d, segDist(x, z, pathPts[i], pathPts[i + 1]));
  return d;
}
function islandR(x, z) { return 24.6 + hash2(Math.round(x / U), Math.round(z / U)) * 2.2; }
function cellH(x, z) {
  if (pondSigned(x, z) < 0) return -2.6;
  if (streamDist(x, z) < 1.8) return -1.7;
  return 0;
}
