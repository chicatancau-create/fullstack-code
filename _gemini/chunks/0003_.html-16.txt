function makeRock(size, mi = 0) {
  const g = new THREE.IcosahedronGeometry(size, 1);
  const p = g.attributes.position;
  for (let i = 0; i < p.count; i++) {
    tmpV.fromBufferAttribute(p, i);
    const n = 1 + 0.42 * fbm(tmpV.x * 5.5, tmpV.z * 5.5 + tmpV.y * 3, 3, 77);
    tmpV.multiplyScalar(n);
    tmpV.y *= rr(0.55, 0.85);
    p.setXYZ(i, tmpV.x, tmpV.y, tmpV.z);
  }
  g.computeVertexNormals();
  const mesh = new THREE.Mesh(g, rockMats[mi % rockMats.length]);
  mesh.castShadow = true; mesh.receiveShadow = true;
  return mesh;
}

function branchCoral(out, pos, dir, len, r0, depth, col) {
  const q = new THREE.Quaternion().setFromUnitVectors(UPV, dir.clone().normalize());
  const g = new THREE.CylinderGeometry(r0 * 0.62, r0, len, 6, 1, depth === 0);
  const m = M().compose(pos, q, new THREE.Vector3(1, 1, 1));
  const tipY = pos.y + len * 0.5;
  out.push(stamp(xform(g, m), col, (x, y) => Math.max(0, (y - pos.y + len * 0.5)) / (len * 1.2) * 0.25));
  if (depth <= 0) return;
  const n = 2 + (rnd() > 0.55 ? 1 : 0);
  for (let i = 0; i < n; i++) {
    const nd = dir.clone()
      .applyAxisAngle(new THREE.Vector3(1, 0, 0.3).normalize(), rr(-0.75, 0.75))
      .applyAxisAngle(UPV, rr(-1.1, 1.1))
      .addScaledVector(UPV, rr(0.15, 0.55)).normalize();
    const np = pos.clone().addScaledVector(dir, len * rr(0.62, 1.0));
    branchCoral(out, np, nd, len * rr(0.52, 0.78), r0 * 0.66, depth - 1, col);
  }
}
