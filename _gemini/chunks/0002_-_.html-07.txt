/* ============ 粉色花树：递归枝干 + 方形花冠 + 根系 ============ */
const treeTops = [];
const BLOSSOM = [0xff8fb8, 0xff6fa5, 0xffa8c8, 0xffc4da, 0xf75590, 0xfff0f5];
function tree(tx, tz, s) {
  const bark = [0x6d4a2f, 0x7d5636, 0x5c3d26];
  function branch(x, y, z, dx, dy, dz, len, w, depth) {
    const steps = Math.max(2, Math.round(len / (w * 0.8)));
    let px = x, py = y, pz = z;
    for (let i = 0; i < steps; i++) {
      vox(px, py, pz, w, w, w, pick(bark), rr(0, 3));
      px += dx * len / steps + rr(-0.06, 0.06);
      py += dy * len / steps;
      pz += dz * len / steps + rr(-0.06, 0.06);
      const j = 0.14;
      dx += rr(-j, j); dz += rr(-j, j); dy += rr(-j * 0.5, j * 0.7);
      const n = Math.hypot(dx, dy, dz); dx /= n; dy /= n; dz /= n;
    }
    if (depth >= 3 || w < 0.24) {
      const R = rr(1.5, 2.1) * s;
      for (let i = 0; i < 30; i++) {
        const a = rr(0, Math.PI * 2), b = rr(-1, 1), r = R * Math.cbrt(rnd());
        const cs = rr(0.42, 0.85) * s;
        vox(px + Math.cos(a) * r * Math.sqrt(1 - b * b), py + b * r * 0.72, pz + Math.sin(a) * r * Math.sqrt(1 - b * b),
          cs, cs * 0.8, cs, rnd() < 0.1 ? 0xfff0f5 : pick(BLOSSOM), rr(0, 3));
      }
      treeTops.push({ x: px, y: py, z: pz, r: R, g: 0 });
      return;
    }
    const kids = depth === 0 ? 3 : (rnd() < 0.5 ? 2 : 3);
    for (let k = 0; k < kids; k++) {
      const a = rr(0, Math.PI * 2), tilt = rr(0.35, 0.85);
      let nx = dx + Math.cos(a) * tilt, nz = dz + Math.sin(a) * tilt, ny = dy + rr(0.25, 0.7);
      const n = Math.hypot(nx, ny, nz);
      branch(px, py, pz, nx / n, ny / n, nz / n, len * rr(0.66, 0.8), w * 0.62, depth + 1);
    }
  }
  branch(tx, 0.35, tz, rr(-0.08, 0.08), 1, rr(-0.08, 0.08), 3.1 * s, 0.85 * s, 0);
  for (let k = 0; k < 6; k++) {
    const a = k / 6 * Math.PI * 2 + rr(-0.2, 0.2);
    for (let i = 0; i < 3; i++) {
      const w = (0.5 - i * 0.13) * s;
      vox(tx + Math.cos(a) * (0.6 + i * 0.55) * s, 0.16 - i * 0.14, tz + Math.sin(a) * (0.6 + i * 0.55) * s,
        w, w * 0.8, w, pick(bark), a);
    }
  }
}
tree(-17, 7, 1.15); tree(3, -16, 1.3); tree(17, -8, 1.0);

      vox(x + lean * k, y + 0.38, z + lean * k * 0.6, 0.3, 0.1, 0.3, 0x2c7a35);
    }
    const ty = baseY + 0.4 + segs * 0.78;
