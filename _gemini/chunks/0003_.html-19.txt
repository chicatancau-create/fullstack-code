        float d = length(gl_PointCoord - 0.5);
        float a = smoothstep(0.5, 0.02, d);
        gl_FragColor = vec4(vec3(0.45,1.0,0.9)*vTw*1.3, a*vTw*uNight*uNight*0.55);
      }`
  });
  const planktonMesh = new THREE.Points(g, mat);
  planktonMesh.frustumCulled = false;
  planktonMesh.renderOrder = 30;
  U.planktonPix = { value: 1 };
  mat.uniforms.uPix = U.planktonPix;
  island.add(planktonMesh);
  plankton = planktonMesh;
}

function presetAt(t) {
  const n = PRESETS.length - 1;
  const x = Math.min(1, Math.max(0, t)) * n;
  const i = Math.min(n - 1, Math.floor(x));
  return { a: PRESETS[i], b: PRESETS[i + 1], f: x - i };
}
let envDirty = true, envSkip = 0;
function applyTime(t) {

  const L = (k) => mix(a[k], b[k], f);
