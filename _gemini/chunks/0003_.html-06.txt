    + 0.030 * fbm(x * 3.4, z * 3.4, 3, 5);

  const hill = 0.14 * Math.exp(-((x - 0.60) ** 2 + (z + 0.30) ** 2) / (2 * 0.26 * 0.26));
  const land = 0.255 + 0.30 * smoothstep(-0.06, 0.66, inland) + hill
    + 0.050 * fbm(x * 2.6 + 7, z * 2.6 - 2, 4, 9)
    + 0.014 * Math.sin(x * 22 + fbm(x * 2, z * 2, 2, 3) * 4);

  let h = mix(seabed, land, t);
  // 靠墙收边：让截面不至于悬空
  h = mix(h, Math.max(h, -0.02), smoothstep(0.93, 1.0, rho) * 0.5);
  return Math.max(h, Y_BOT + 0.03);
}
function terrainN(x, z, out = new THREE.Vector3()) {
  const e = 0.006;
  const dx = (terrainH(x + e, z) - terrainH(x - e, z)) / (2 * e);
  const dz = (terrainH(x, z + e) - terrainH(x, z - e)) / (2 * e);
  return out.set(-dx, 1, -dz).normalize();
}
