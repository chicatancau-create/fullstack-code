const HOSTS = [
  'https://fastly.jsdelivr.net/npm/three@0.169.0/',
  'https://cdn.jsdelivr.net/npm/three@0.169.0/',
  'https://unpkg.com/three@0.169.0/',
  'https://registry.npmmirror.com/three/0.169.0/files/'
];
async function imp(spec) {
  try { return await import(spec); }
  catch (e) {
    const rel = spec.startsWith('three/addons/') ? 'examples/jsm/' + spec.slice('three/addons/'.length) : null;
    if (!rel) throw e;                       // 核心库必须与 importmap 同源，换源会得到两份 three
    let last = e;
    for (const h of HOSTS) { try { return await import(h + rel); } catch (e2) { last = e2; } }
    throw last;
  }
}
