function freeSpot(x, z, pad = 0.6) {
  if (Math.hypot(x, z) > 23.2) return false;
  if (cellH(x, z) !== 0) return false;
  if (pathDist(x, z) < 1.9) return false;
  for (const o of reserved) if (Math.hypot(x - o.x, z - o.z) < o.r + pad) return false;
  return true;
}
