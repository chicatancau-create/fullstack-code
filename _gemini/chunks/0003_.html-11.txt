const shellUni = Object.assign(skyUniformValues(), {
  uPre: U.pre, uPreDepth: U.preDepth, uProjInv: U.projInv,
  uCamWorld: U.camWorld, uViewProj: U.viewProj,

const shellMat = new THREE.ShaderMaterial({
  uniforms: shellUni, transparent: true, blending: THREE.NoBlending,
  depthWrite: false, side: THREE.FrontSide
});
shellMat.vertexShader = /* glsl */`
  varying vec3 vWPos; varying vec3 vNrm; varying vec2 vSUV;
  void main(){
    vec4 wp = modelMatrix*vec4(position,1.0);
    vWPos = wp.xyz;
    vNrm = normalize( mat3(modelMatrix)*normal );
    vec4 cp = projectionMatrix*viewMatrix*wp;
    vSUV = cp.xy/cp.w*0.5+0.5;
    gl_Position = cp;
  }`;
shellMat.fragmentShader = GLSL_NOISE + GLSL_SAFE + GLSL_SKY + /* glsl */`
  uniform sampler2D uPre, uPreDepth;
  uniform mat4 uProjInv, uCamWorld, uViewProj;
  uniform vec3 uShellPos, uResinAbsorb, uResinTint;
  uniform float uRi, uIOR, uDisp, uScratch, uRefl, uTime, uWaterY;
  varying vec3 vWPos; varying vec3 vNrm; varying vec2 vSUV;

  // 与以 Y 为轴、半径 rad 的圆柱求交（取正向最近交点）
  float cylT(vec3 o, vec3 d, float rad, out vec3 hitN){
    float a = dot(d.xz,d.xz);
    float b = 2.0*dot(o.xz,d.xz);
    float c = dot(o.xz,o.xz) - rad*rad;
    float disc = b*b - 4.0*a*c;
    if ( disc <= 0.0 || a < 1e-6 ) { hitN = vec3(0.0); return -1.0; }
    float sq = sqrt(disc);
    float t1 = (-b - sq)/(2.0*a);
    float t2r = (-b + sq)/(2.0*a);
    float t = t1 > 1e-4 ? t1 : (t2r > 1e-4 ? t2r : -1.0);
    if ( t < 0.0 || t > 0.6 ) { hitN = vec3(0.0); return -1.0; }
    vec3 hp = o + d*t;
    float hl = length(hp.xz);
    hitN = hl > 1e-4 ? vec3(hp.x, 0.0, hp.z)/hl : vec3(0.0, 0.0, 1.0);
    hitN *= ( dot(d, hitN) > 0.0 ? -1.0 : 1.0 );
    return t;
  }

  vec2 projSafe(vec3 P, vec2 fallback){
    vec4 c = uViewProj*vec4(P,1.0);
    if ( c.w <= 1e-4 ) return fallback;
    return clamp((c.xy/c.w)*0.5+0.5, 0.0, 1.0);
  }

  void main(){
    vec3 V = normalize( vWPos - cameraPosition );
    vec3 N = normalize( vNrm );

    // 树脂表面微划痕/指纹扰动
    vec3 lp = vWPos - uShellPos;
    float g1 = vn( lp.xz*13.0 + vec2(lp.y*4.0, -lp.y*3.0) );
    float g2 = vn( lp.zx*9.0 + vec2(13.0, 7.0) + lp.y );
    N = normalize( N + vec3(g1-0.5, g2-0.5, vn(lp.xy*55.0)-0.5)*0.018*uScratch );

    // 1) 空气→树脂
