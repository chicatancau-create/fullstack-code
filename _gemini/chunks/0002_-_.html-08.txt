function paversAlong(pts, step, size, color, y = 0.05) {
  for (let i = 0; i < pts.length - 1; i++) {
    const a = pts[i], b = pts[i + 1];
    const len = Math.hypot(b[0] - a[0], b[1] - a[1]);
    const n = Math.floor(len / step);
    for (let k = 0; k <= n; k++) {
      const t = k / Math.max(1, n);
      const x = a[0] + (b[0] - a[0]) * t + rr(-0.12, 0.12);
      const z = a[1] + (b[1] - a[1]) * t + rr(-0.12, 0.12);
      if (cellH(x, z) !== 0) continue;
      vox(x, y, z, size * rr(0.85, 1.05), 0.09, size * rr(0.85, 1.05), color, rr(-0.2, 0.2));
    }
  }
}
