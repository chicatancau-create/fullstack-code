    <h2>4. 汇总：reduce / find / some / every / sort</h2>
    <pre><code>const nums = [3, 1, 4, 1, 5];
console.log('求和', nums.reduce((a, b) =&gt; a + b, 0));
console.log('最大值', nums.reduce((a, b) =&gt; (a &gt; b ? a : b)));
console.log('find 第一个&gt;2', nums.find((v) =&gt; v &gt; 2));
console.log('some 有&gt;4?', nums.some((v) =&gt; v &gt; 4), 'every 都&gt;0?', nums.every((v) =&gt; v &gt; 0));

    <h2>5. 对象数组（表格数据的来源）</h2>
    <pre><code>const goods = [
  { name: '女鞋', price: 298, tags: ['舒适', '透气'] },
  { name: '雪地靴', price: 89, tags: ['保暖', '防滑'] },
  { name: '毛衣', price: 199, tags: ['秋冬'] },
];
console.log(goods.map((g) =&gt; g.name));
console.log('总价', goods.reduce((s, g) =&gt; s + g.price, 0));
console.log('500 以下', goods.filter((g) =&gt; g.price &lt; 500).length, '件');
console.log('所有标签', goods.flatMap((g) =&gt; g.tags));</code></pre>
